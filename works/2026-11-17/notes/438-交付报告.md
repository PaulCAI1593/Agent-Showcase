```python
#!/usr/bin/env python3
# 复刻 claim_gate.py 的 G1 校验逻辑，对单件作品的 claims JSON 做逐字回源
import json, re, sys, os

PUNCT = '，。、；：""''（）()·—…!！?？「」《》'

def normalize(s):
    s = re.sub(r'<[^>]+>', ' ', s)
    s = re.sub(r'\s+', '', s)
    for c in PUNCT:
        s = s.replace(c, '')
    return s

def strip_code(html):
    s = re.sub(r'<!--.*?-->', ' ', html, flags=re.S)
    s = re.sub(r'<(script|style)\b.*?</\1>', ' ', s, flags=re.S | re.I)
    return s

path = sys.argv[1]
raw = open(path, encoding='utf-8').read()
body = normalize(strip_code(raw))

m = re.search(r'<script type="application/json" id="claims">(.*?)</script>', raw, re.S)
claims = json.loads(m.group(1))

print(f'文件: {path}')
print(f'claims 条数: {len(claims)}')
print(f'归一化正文长度: {len(body)} 字符')
print()
print('idx | label | value | normalize(quote) | 命中')
print('-' * 100)
ok = 0
for i, c in enumerate(claims, 1):
    q = normalize(c['quote'])
    hit = q in body
    if hit:
        ok += 1
    print(f'{i:>3} | {c["label"]} | {c["value"]} | {q} | {"命中" if hit else "未命中"}')
print('-' * 100)
print(f'逐字命中: {ok}/{len(claims)}')
```
